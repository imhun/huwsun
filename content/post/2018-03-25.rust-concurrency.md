并发

<!-- more -->

共享可变，convar，与一个mutex保护的条件量一起起作用

wait方法会一直等待，直到子线程调用notify方法

wait方法接受一个MutexGuard值，消耗掉，并释放锁，然后等待子线程的notify通知，等到通知以后，再次lock并返回一个MutexGuard值

```rust
use std::sync::{Arc, Mutex, Condvar};
use std::thread;
fn main() {
let pair = Arc::new((Mutex::new(false), Condvar::new()));
let pair2 = pair.clone();
// 创建一个新线程
thread::spawn(move|| {
let &(ref lock, ref cvar) = &*pair2;
println!("sub thread sleeping");
thread::sleep_ms(100);
println!("sub thread locking");
let mut started = lock.lock().unwrap();
println!("sub thread locked");
*started = true;
println!("sub notify");
cvar.notify_one();//不调用该方法，主线程会一直阻塞
println!("after notify sleeping");
thread::sleep_ms(10000);
println!("notify main thread");
});
// 等待新线程先运行
let &(ref lock, ref cvar) = &*pair;
println!("main thread sleep");
//thread::sleep_ms(2000);
println!("main thread locking");
let mut started = lock.lock().unwrap();
println!("main thread locked");
while !*started {
println!("before wait");
started = cvar.wait(started).unwrap();//wait方法在子线程nofiy，并且lock释放以后重新获取lock
println!("after wait");
}
}
```
